function startStudio() {

  document.getElementById("intro").style.display = "none";

  document

}


function goTo(id) {

  document
    .querySelectorAll(".screen")
    .forEach(screen => {
      screen.classList.remove("active");
    });

  const target =
    document.getElementById(id);

  if (target) {
    target.classList.add("active");
  }

  window.scrollTo({
    top: 0,
    behavior: "smooth"
  });
}

/* =========================
   CHARACTER SYSTEM
========================= */

let characters = [];

let editingCharacterId = null;


/* GENERATE CHARACTER */

function generateCharacter() {

  const name =
    document
      .getElementById("characterName")
      .value
      .trim();

  const description =
    document
      .getElementById(
        "characterDescription"
      )
      .value
      .trim();

  const style =
    document
      .getElementById(
        "characterStyle"
      )
      .value;


  if (!name) {

    alert(
      "Please enter a character name."
    );

    return;
  }


  if (!description) {

    alert(
      "Please describe the character."
    );

    return;
  }


  /*
    This prototype creates a local
    illustrated character preview.

    Later this function can be connected
    to a real AI image-generation API.
  */

  const avatar =
    createCharacterAvatar(
      name,
      style
    );


  document
    .getElementById(
      "characterPreview"
    )
    .innerHTML = `

      <div class="generated-character">

        <div
          class="character-avatar"
          style="
            background:
            linear-gradient(
              145deg,
              ${avatar.color1},
              ${avatar.color2}
            );
          "
        >
          ${avatar.emoji}
        </div>

        <div class="character-name-preview">
          ${escapeHTML(name)}
        </div>

        <div class="character-style-preview">
          ${escapeHTML(style)}
        </div>

      </div>

    `;


  if (editingCharacterId !== null) {

    updateCharacter(
      editingCharacterId,
      name,
      description,
      style,
      avatar
    );

    editingCharacterId = null;

  } else {

    addCharacterToCollection(
      name,
      description,
      style,
      avatar
    );

  }

  updateCharacterCount();
}


/* =========================
   CREATE LOCAL CHARACTER ART
========================= */

function createCharacterAvatar(
  name,
  style
) {

  const styles = {

    comic: {
      color1: "#7040e8",
      color2: "#1ab8d2",
      emoji: "🦸"
    },

    anime: {
      color1: "#ff5c9a",
      color2: "#7040e8",
      emoji: "⚔️"
    },

    cartoon: {
      color1: "#ff9d42",
      color2: "#ffd34e",
      emoji: "😎"
    },

    manga: {
      color1: "#555",
      color2: "#151515",
      emoji: "🗡️"
    },

    cinematic: {
      color1: "#172b4d",
      color2: "#7c3aed",
      emoji: "🎬"
    }

  };


  return (
    styles[style] ||
    styles.comic
  );
}


/* =========================
   ADD CHARACTER
========================= */

function addCharacterToCollection(
  name,
  description,
  style,
  avatar
) {

  const character = {

    id: Date.now(),

    name,

    description,

    style,

    avatar

  };


  characters.push(character);

  renderCharacters();
}


/* =========================
   DISPLAY ALL CHARACTERS
========================= */

function renderCharacters() {

  const list =
    document.getElementById(
      "characterList"
    );


  list.innerHTML = "";


  characters.forEach(character => {

    const card =
      document.createElement("div");


    card.className =
      "character-card";


    card.innerHTML = `

      <div
        class="character-card-image"
      >

        <div
          class="card-avatar"
          style="
            background:
            linear-gradient(
              145deg,
              ${character.avatar.color1},
              ${character.avatar.color2}
            );
          "
        >
          ${character.avatar.emoji}
        </div>

      </div>


      <div class="character-card-info">

        <h3>
          ${escapeHTML(character.name)}
        </h3>

        <p>
          ${escapeHTML(
            character.description
          )}
        </p>

        <small>
          🎨 ${escapeHTML(
            character.style
          )}
        </small>


        <div class="card-buttons">

          <button
            onclick="
              editCharacter(${character.id})
            "
          >
            ✏️ Edit
          </button>

          <button
            onclick="
              regenerateCharacter(
                ${character.id}
              )
            "
          >
            🔄 Regenerate
          </button>

          <button
            onclick="
              deleteCharacter(
                ${character.id}
              )
            "
          >
            🗑️
          </button>

        </div>

      </div>

    `;


    list.appendChild(card);

  });


  updateCharacterCount();
}


/* =========================
   EDIT CHARACTER
========================= */

function editCharacter(id) {

  const character =
    characters.find(
      item => item.id === id
    );


  if (!character) {
    return;
  }


  document
    .getElementById(
      "characterName"
    )
    .value =
    character.name;


  document
    .getElementById(
      "characterDescription"
    )
    .value =
    character.description;


  document
    .getElementById(
      "characterStyle"
    )
    .value =
    character.style;


  editingCharacterId = id;


  window.scrollTo({
    top: 0,
    behavior: "smooth"
  });


  document
    .getElementById(
      "characterName"
    )
    .focus();
}


/* =========================
   UPDATE CHARACTER
========================= */

function updateCharacter(
  id,
  name,
  description,
  style,
  avatar
) {

  const character =
    characters.find(
      item => item.id === id
    );


  if (!character) {
    return;
  }


  character.name =
    name;

  character.description =
    description;

  character.style =
    style;

  character.avatar =
    avatar;


  renderCharacters();
}


/* =========================
   REGENERATE
========================= */

function regenerateCharacter(id) {

  const character =
    characters.find(
      item => item.id === id
    );


  if (!character) {
    return;
  }


  character.avatar =
    createCharacterAvatar(
      character.name,
      character.style
    );


  renderCharacters();
}


/* =========================
   DELETE
========================= */

function deleteCharacter(id) {

  const confirmed =
    confirm(
      "Delete this character?"
    );


  if (!confirmed) {
    return;
  }


  characters =
    characters.filter(
      character =>
        character.id !== id
    );


  renderCharacters();
}

/* =========================
   STEP 4
   PANELS
========================= */

function selectPanel(number) {

  const preview =
    document.getElementById(
      "panelPreview"
    );


  preview.style.gridTemplateColumns =
    number <= 2
      ? `repeat(${number}, 1fr)`
      : "repeat(2, 1fr)";


  preview.innerHTML = "";


  for (
    let i = 1;
    i <= number;
    i++
  ) {

    const panel =
      document.createElement(
        "div"
      );


    panel.style.background =
      "#d5d5d5";

    panel.style.border =
      "4px solid #111";

    panel.style.minHeight =
      "120px";

    panel.style.display =
      "flex";

    panel.style.alignItems =
      "center";

    panel.style.justifyContent =
      "center";


    panel.innerHTML =
      `<strong>
        Panel ${i}
      </strong>`;


    preview.appendChild(
      panel
    );
  }


  const pagePanels =
    document.querySelector(
      ".page-panels"
    );


  pagePanels.innerHTML = "";


  for (
    let i = 1;
    i <= number;
    i++
  ) {

    const panel =
      document.createElement(
        "div"
      );


    panel.className =
      "page-panel";


    panel.innerHTML =
      `Panel ${i}`;


    pagePanels.appendChild(
      panel
    );
  }
}


/* =========================
   DOWNLOAD
========================= */

function downloadComic() {

  const content =
    document
      .querySelector(
        ".comic-page"
      )
      .outerHTML;


  const blob =
    new Blob(

      [`

        <html>

        <head>

        <title>
          My Comic
        </title>

        <style>

        body {
          background:#222;
          padding:30px;
        }

        .comic-page {
          background:white;
          width:600px;
          min-height:700px;
          padding:20px;
          margin:auto;
        }

        .page-panels {
          display:grid;
          grid-template-columns:
          repeat(2,1fr);
          gap:10px;
        }

        .page-panel {
          min-height:170px;
          background:#ddd;
          border:4px solid #111;
        }

        </style>

        </head>

        <body>

        ${content}

        </body>

        </html>

      `],

      {
        type:
          "text/html"
      }

    );


  const url =
    URL.createObjectURL(
      blob
    );


  const a =
    document.createElement(
      "a"
    );


  a.href = url;

  a.download =
    "my-comic.html";


  a.click();


  URL.revokeObjectURL(
    url
  );
}

/* =========================
   SECURITY
========================= */

function escapeHTML(text) {

  const div =
    document.createElement(
      "div"
    );


  div.textContent =
    text;


  return div.innerHTML;
}
